# 7. API Integration Layer

## api.ts

```typescript
const API_BASE_URL = "https://localhost:5001/api";

export async function apiFetch<T>(
  endpoint: string,
  options?: RequestInit
): Promise<T> {
  const response = await fetch(`${API_BASE_URL}${endpoint}`, {
    headers: {
      "Content-Type": "application/json",
      ...(options?.headers || {}),
    },
    ...options,
  });

  if (!response.ok) {
    const error = await response.text();
    throw new Error(error || "Something went wrong");
  }

  return response.json();
}
```

---

## warrantyService.ts

```typescript
// src/services/warrantyService.ts

import { apiFetch } from "./api";
import {
  Device,
  PurchaseSource,
  WarrantyRegistration,
  WarrantyResult,
} from "../types/warranty";

export const warrantyService = {
  getDevices: () =>
    apiFetch<Device[]>("/devices"),

  getPurchaseSources: () =>
    apiFetch<PurchaseSource[]>("/purchasesources"),

  searchWarrantyByMobile: (mobileNumber: string) =>
    apiFetch<WarrantyResult[]>(
      `/warrantyregistrations/search/${mobileNumber}`
    ),
};
```


## warrantyService.ts

```typescript

const handleSubmit = async (data: WarrantyRegistrationForm) => {
  const formData = new FormData();

  formData.append("OwnerName", data.ownerName);
  formData.append("EmailAddress", data.emailAddress ?? "");
  formData.append("MobileNumber", data.mobileNumber);
  formData.append("DeviceId", data.deviceId.toString());
  formData.append(
    "PurchaseSourceId",
    data.purchaseSourceId.toString()
  );
  formData.append("PurchaseDate", data.purchaseDate);
  formData.append("Notes", data.notes ?? "");

  if (data.invoiceFile?.[0]) {
    formData.append(
      "InvoiceFile",
      data.invoiceFile[0]
    );
  }

  const response = await fetch(
	  "https://localhost:7045/api/WarrantyRegistrations",
	  {
		method: "POST",
		headers: {
		  Accept: "application/json",
		},
		body: formData,
	  }
	);

  if (!response.ok) {
    throw new Error("Failed to register warranty");
  }

  const result = await response.json();

  console.log(result);
};

```

## Example Usage in React Component

```typescript

import { useEffect, useState } from "react";
import { warrantyService } from "../services/warrantyService";

function DeviceDropdown() {
  const [devices, setDevices] = useState([]);

  useEffect(() => {
    const loadDevices = async () => {
      try {
        const data = await warrantyService.getDevices();
        setDevices(data);
      } catch (error) {
        console.error(error);
      }
    };

    loadDevices();
  }, []);

  return (
    <select>
      {devices.map((device: any) => (
        <option key={device.id} value={device.id}>
          {device.name}
        </option>
      ))}
    </select>
  );
}

export default DeviceDropdown;

```
